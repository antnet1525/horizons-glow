        function createTulips() {
            const positions = [
                { top: '5%', left: '3%', size: 140, rotation: 15 },
                { top: '10%', right: '5%', size: 160, rotation: -20 },
                { top: '40%', left: '2%', size: 150, rotation: 35 },
                { top: '50%', right: '4%', size: 145, rotation: -25 },
                { bottom: '10%', left: '6%', size: 155, rotation: 45 },
                { bottom: '15%', right: '3%', size: 165, rotation: -15 },
                { top: '25%', left: '8%', size: 135, rotation: 60 },
                { bottom: '35%', right: '7%', size: 140, rotation: -40 }
            ];

            positions.forEach(pos => {
                const svg = document.createElementNS('http://www.w3.org/2000/svg', 'svg');
                svg.classList.add('tulip');
                svg.setAttribute('viewBox', '0 0 100 120');
                svg.setAttribute('width', pos.size);
                svg.setAttribute('height', pos.size * 1.2);
                svg.style.transform = `rotate(${pos.rotation}deg)`;
                
                if (pos.top) svg.style.top = pos.top;
                if (pos.bottom) svg.style.bottom = pos.bottom;
                if (pos.left) svg.style.left = pos.left;
                if (pos.right) svg.style.right = pos.right;

                const g = document.createElementNS('http://www.w3.org/2000/svg', 'g');
                g.setAttribute('transform', 'translate(50, 40)');

                // Outer petals (3 back petals)
                const outerPetals = [
                    { d: 'M 0,-25 Q -15,-35 -20,-15 Q -15,5 0,0 Z', fill: '#E8A3B0' },
                    { d: 'M 0,-25 Q 0,-40 0,-15 Q 0,5 0,0 Z', fill: '#F1B7C2' },
                    { d: 'M 0,-25 Q 15,-35 20,-15 Q 15,5 0,0 Z', fill: '#E8A3B0' }
                ];

                outerPetals.forEach(petal => {
                    const path = document.createElementNS('http://www.w3.org/2000/svg', 'path');
                    path.setAttribute('d', petal.d);
                    path.setAttribute('fill', petal.fill);
                    path.setAttribute('opacity', '0.9');
                    g.appendChild(path);
                });

                // Inner petals (2 front petals)
                const innerPetals = [
                    { d: 'M 0,0 Q -18,-5 -12,-20 Q -5,-10 0,-15 Z', fill: '#F8CCD7' },
                    { d: 'M 0,0 Q 18,-5 12,-20 Q 5,-10 0,-15 Z', fill: '#F8CCD7' }
                ];

                innerPetals.forEach(petal => {
                    const path = document.createElementNS('http://www.w3.org/2000/svg', 'path');
                    path.setAttribute('d', petal.d);
                    path.setAttribute('fill', petal.fill);
                    path.setAttribute('opacity', '0.95');
                    g.appendChild(path);
                });

                // Stem
                const stem = document.createElementNS('http://www.w3.org/2000/svg', 'path');
                stem.setAttribute('d', 'M 0,0 Q -2,20 0,45');
                stem.setAttribute('stroke', '#7A9E7E');
                stem.setAttribute('stroke-width', '2.5');
                stem.setAttribute('fill', 'none');
                stem.setAttribute('opacity', '0.7');
                g.appendChild(stem);

                svg.appendChild(g);
                document.body.appendChild(svg);
            });
        }

        createTulips();
    </script>
